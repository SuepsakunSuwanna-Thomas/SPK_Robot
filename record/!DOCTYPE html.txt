        function validateCustomBlockCode(code) {
            if (!code || typeof code !== 'string') throw new Error('Invalid code input');
            const dangerousPatterns = [/eval\s*\(/gi, /exec\s*\(/gi, /system\s*\(/gi, /__.*__/gi, /process\./gi, /require\s*\(/gi, /import\s+/gi];
            for (let pattern of dangerousPatterns) {
                if (pattern.test(code)) throw new Error('Dangerous code pattern detected');
            }
            if (code.length > 5000) throw new Error('Code too large (max 5000 characters)');
            return true;
        }

        function cleanUpWorkspace() {
            const topBlocks = workspace.getTopBlocks(true);
            let setupBlock = null;
            let loopBlock = null;
            const sideBlocks = [];

            topBlocks.forEach(b => {
                if (b.type === 'arduino_setup') setupBlock = b;

                else sideBlocks.push(b);
            });

            if (!setupBlock || !loopBlock) {
                workspace.cleanUp();
                workspace.scrollCenter();
                return;
            }

            // 🎯 CENTERED SPINE: Setup and Loop in middle of view
            const centerX = 2000;
            const centerY = 2000;

            // Position Setup above center
            setupBlock.moveBy(centerX - setupBlock.getRelativeToSurfaceXY().x, (centerY - 250) - setupBlock.getRelativeToSurfaceXY().y);

            // Position Loop below Setup
            const setupSize = setupBlock.getHeightWidth();
            const loopY = (centerY - 250) + setupSize.height + 150;
            loopBlock.moveBy(centerX - loopBlock.getRelativeToSurfaceXY().x, loopY - loopBlock.getRelativeToSurfaceXY().y);

            // Branch others symmetrically
            const leftBlocks = sideBlocks.filter((_, i) => i % 2 === 0);
            const rightBlocks = sideBlocks.filter((_, i) => i % 2 !== 0);

            let leftY = centerY - 250;
            leftBlocks.forEach(b => {
                b.moveBy((centerX - 600) - b.getRelativeToSurfaceXY().x, leftY - b.getRelativeToSurfaceXY().y);
                leftY += b.getHeightWidth().height + 60;
            });

            let rightY = centerY - 250;
            rightBlocks.forEach(b => {
                b.moveBy((centerX + 600) - b.getRelativeToSurfaceXY().x, rightY - b.getRelativeToSurfaceXY().y);
                rightY += b.getHeightWidth().height + 60;
            });

            // 🎯 CENTER VIEW on the Spine
            workspace.scrollCenter();
            showStatusToast("จัดระเบียบและวาง Setup & Loop ไว้กลางจอแล้ว");

        async function parseAndAddCustomBlock() {
            let code = document.getElementById('cbCode').value.trim();
            let category = document.getElementById('cbCategory').value.trim() || "My Blocks";
            try { validateCustomBlockCode(code); } catch (error) { return Swal.fire("Security Error", error.message, "error"); }

            let match = code.match(/void\s+([a-zA-Z0-9_]+)\s*\(([^)]*)\)\s*\{([\s\S]*)\}/);
            if (!match) return Swal.fire("รูปแบบผิดพลาด", "โค้ดต้องเป็นฟังก์ชัน void (...) { ... }", "error");

            let name = match[1]; let argsRaw = match[2]; let args = [];

            let existingIndex = window.customBlocksData.findIndex(cb => cb.name === name);
            if (existingIndex > -1 && window.editingBlockId === null) {
                const { isConfirmed } = await Swal.fire({ title: 'ชื่อซ้ำ', text: `เขียนทับ ${name}?`, showCancelButton: true, confirmButtonText: 'เขียนทับ' });
                if (!isConfirmed) return;
                window.editingBlockId = window.customBlocksData[existingIndex].id;
            }

            if (argsRaw.trim() !== "") {
                let argParts = argsRaw.split(','); let htmlInputs = '';
                argParts.forEach((part, index) => {
                    let words = part.trim().split(/\s+/); let argName = words.pop(); let argType = words.join(' ');
                    if (!validateArgType(argType)) throw new Error(`Invalid arg: ${argType}`);
                    let existVal = 0;
                    if (window.editingBlockId) { let cb = window.customBlocksData.find(b => b.id === window.editingBlockId); if (cb && cb.args && cb.args[index]) existVal = cb.args[index].defaultVal || 0; }
                    htmlInputs += `<div style="text-align:left; margin-bottom:12px;"><label style="font-size:13px; font-weight:600; color:var(--text-main); margin-bottom:4px; display:block;">${escapeHtml(argName)}</label><input type="text" id="argDef_${index}" class="swal2-input" value="${escapeHtml(String(existVal))}" style="margin:0; width:100%;"></div>`;
                    args.push({ name: argName, type: argType, defaultVal: existVal });
                });
                if (args.length > 0) {
                    const { isConfirmed } = await Swal.fire({
                        title: 'ตั้งค่าเริ่มต้น', html: htmlInputs, confirmButtonText: 'บันทึก', showCancelButton: true,
                        preConfirm: () => { args.forEach((arg, index) => { arg.defaultVal = document.getElementById(`argDef_${index}`).value || 0; }); }
                    });
                    if (!isConfirmed) return;
                }
            }

            if (window.editingBlockId) {
                let idx = window.customBlocksData.findIndex(cb => cb.id === window.editingBlockId);
                if (idx > -1) window.customBlocksData[idx] = { id: window.editingBlockId, name, category, args, code };
                window.editingBlockId = null;
            } else { window.customBlocksData.push({ id: Date.now().toString(), name, category, args, code }); }

            reloadCustomBlocks();
            let currentXml = Blockly.Xml.domToText(Blockly.Xml.workspaceToDom(workspace));
            workspace.clear();
            Blockly.Xml.domToWorkspace(Blockly.utils.xml.textToDom(currentXml), workspace);
            openCustomBlockManager();
        }

        function updateSearchCount() {
            if (!window.srcAceEditor) return;
            const findStr = document.getElementById('srcFind').value;
            const countEl = document.getElementById('srcMatchCount');
            if (!findStr) { countEl.innerText = '0 / 0'; return; }
            const Search = ace.require("ace/search").Search;
            const search = new Search();
            search.setOptions({ needle: findStr, caseSensitive: false, wrap: true });
            const allRanges = search.findAll(window.srcAceEditor.session);
            let currentIndex = 0;
            const currentSel = window.srcAceEditor.getSelectionRange();
            for (let i = 0; i < allRanges.length; i++) {
                if (allRanges[i].start.row === currentSel.start.row && allRanges[i].start.column === currentSel.start.column) {
                    currentIndex = i + 1;
                    break;
                }
            }
            countEl.innerText = `${currentIndex || (allRanges.length > 0 ? 1 : 0)} / ${allRanges.length}`;
        }

        function srcFindText(e) {
            const query = document.getElementById('srcFind').value;
            if (!query) { window.srcAceEditor.find(''); document.getElementById('srcMatchCount').innerText = '0 / 0'; return; }
            window.srcAceEditor.find(query, { backwards: false, wrap: true, caseSensitive: false });
            updateSearchCount();

        async function handleUpload() {
            if (!window.currentProjectName) {

            localStorage.setItem('rp_last_file', filename);

            let wasConnected = false;
            if (window.port) { wasConnected = true; await disconnectSerial(); }

            let uploadAbortController = new AbortController(); let isCancelled = false; let startTime = Date.now();

                    isCancelled = true; uploadAbortController.abort();

                }
            });

            if (isCancelled) return;

            try {
                const res = await fetch('/upload', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ code: window.generatedCppCode, userCode: window.generatedUserCode }), signal: uploadAbortController.signal });

        }

        async function loadFile(n) {
            try {
                const res = await fetch(`/load_file/${encodeURIComponent(n)}`);
                if (!res.ok) throw new Error("File not found");
                const data = await res.json();

                workspace.clear();
                window.customBlocksData = data.customBlocks || []; reloadCustomBlocks();

                try {
                    Blockly.Xml.domToWorkspace(Blockly.utils.xml.textToDom(data.xml), workspace);
                } catch (xmlError) {
                    console.error("Corrupted project XML", xmlError);
                    Swal.fire('ไฟล์เสียหาย', 'ไม่สามารถอ่านโครงสร้างบล็อกได้ ข้อมูลอาจสูญหายบางส่วน', 'warning');
                    Blockly.Xml.domToWorkspace(Blockly.utils.xml.textToDom(defaultWorkspaceXML), workspace);
                }

                window.currentProjectName = n.replace('.rp2350', '');
                localStorage.setItem('rp_last_file', n);
                updateBadge(); Swal.close();
                showStatusToast("โหลดโปรเจกต์สำเร็จ");
                return true;
            } catch (e) {
                if (window.loadingLastFile) {
                    localStorage.removeItem('rp_last_file');
                    window.currentProjectName = "";
                    updateBadge();
                    newProject();
                } else {
                    Swal.fire('เกิดข้อผิดพลาด', 'ไม่สามารถโหลดไฟล์ได้', 'error');
                    Blockly.Xml.domToWorkspace(Blockly.utils.xml.textToDom(defaultWorkspaceXML), workspace);
                }
                return false;
            } finally {
                window.loadingLastFile = false;
            }
        }

        navigator.serial.addEventListener("disconnect", (event) => {

        async function attemptReconnect() {
            if (window.intentionalDisconnect) return;
            const ports = await navigator.serial.getPorts();
            if (ports.length > 0) { try { window.port = ports[0]; await connectPort(); } catch (e) { setTimeout(attemptReconnect, 2000); } }
            else { setTimeout(attemptReconnect, 2000); }
        }

        async function toggleSerial() {
            if (window.port) { window.intentionalDisconnect = true; await disconnectSerial(); }
            else { try { window.port = await navigator.serial.requestPort(); window.intentionalDisconnect = false; await connectPort(); } catch (e) { } }
        }

        async function connectPort() {
            await window.port.open({ baudRate: 115200 });

            // Update Top button
            let btn = document.getElementById('btnSerial');
            btn.innerHTML = "<i class='fa-solid fa-plug-circle-xmark'></i> Disconnect Serial";
            btn.classList.add('danger'); btn.classList.remove('primary');

            // Update Bottom Status bar
            let statusBtn = document.getElementById('bottomSerialStatus');
            let statusText = document.getElementById('bottomSerialText');
            statusBtn.classList.remove('disconnected');
            statusBtn.classList.add('connected');
            statusText.innerText = "Connected (115200)";
            statusBtn.querySelector('i').className = "fa-solid fa-plug-circle-check";

            window.keepReading = true; readLoop();
        }

        async function readLoop() {
            const decoder = new TextDecoderStream();
            window.closedPromise = window.port.readable.pipeTo(decoder.writable);
            window.reader = decoder.readable.getReader();
            const logElement = document.getElementById('serialLog');

            try {
                while (window.keepReading) {
                    const { value, done } = await window.reader.read();
                    if (done) break;

                    // Optimization: Use textContent append or text nodes for performance
                    const span = document.createElement('span');
                    span.textContent = value;
                    logElement.appendChild(span);

                    // Limit total log elements to prevent slow-down over time
                    if (logElement.childNodes.length > 1000) {
                        logElement.removeChild(logElement.firstChild);
                    }

                    if (window.autoScroll) logElement.scrollTop = logElement.scrollHeight;
                }
            } catch (e) { } finally { if (window.reader) window.reader.releaseLock(); }
        }

        function disconnectSerialUI() {
            let btn = document.getElementById('btnSerial');
            btn.innerHTML = "<i class='fa-solid fa-plug'></i> Connect Serial";
            btn.classList.remove('danger'); btn.classList.add('primary');

            // Update Bottom Status bar
            let statusBtn = document.getElementById('bottomSerialStatus');
            let statusText = document.getElementById('bottomSerialText');
            statusBtn.classList.add('disconnected');
            statusBtn.classList.remove('connected');
            statusText.innerText = "No Port Connected";
            statusBtn.querySelector('i').className = "fa-solid fa-circle-xmark";
        }

        async function disconnectSerial() {
            window.keepReading = false; window.intentionalDisconnect = true;
            if (window.reader) { try { await window.reader.cancel(); } catch (e) { } }
            if (window.closedPromise) { try { await window.closedPromise.catch(() => { }); } catch (e) { } }
            if (window.port) { try { await window.port.close(); } catch (e) { } window.port = null; }
            disconnectSerialUI();
        }

        function toggleAutoScroll() {
            window.autoScroll = !window.autoScroll;
            const btn = document.getElementById('btnScroll');
            if (window.autoScroll) { btn.classList.add('primary'); btn.innerHTML = '<i class="fa-solid fa-arrow-down"></i>'; }
            else { btn.classList.remove('primary'); btn.innerHTML = '<i class="fa-solid fa-pause"></i>'; }
        }
        function clearSerial() { document.getElementById('serialLog').innerText = ""; }
        async function sendSerial() {
            if (!window.port) return;
            const val = document.getElementById('serialIn').value + '\n';
            const writer = window.port.writable.getWriter();
            await writer.write(new TextEncoder().encode(val));
            writer.releaseLock();
            document.getElementById('serialIn').value = '';
        }
